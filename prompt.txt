@app.route('/v1/chat/completions', methods=['POST'])
def chat_completions():
    req_data = request.json
    
    # --- 1. Dynamic Payload Construction ---
    payload = {

        "system_prompt": req_data.get("system_prompt", None)
    }

    # Codecraft prefers System Prompts at the root level.
    # If OmniRoute sends it inside the messages array, we extract it.
    if payload["messages"] and payload["messages"][0].get("role") == "system":
        payload["system_prompt"] = payload["messages"][0].get("content")

    # Forward Advanced Features: Tools (for Web Search/Funcs) and JSON Format.
    # Vision (Images) is natively forwarded because it resides inside `messages`.
    for key in ["tools", "tool_choice", "response_format"]:
        if key in req_data:
            payload[key] = req_data[key]

    headers = {
        "accept": "application/json",
        "content-type": "application/json",
        "referer": "https://codecraftapi.com/dashboard/playground",
        "x-xsrf-token": xsrf_token,
        "x-requested-with": "XMLHttpRequest"
    }

    def generate():
        response = active_session.post(
            "https://codecraftapi.com/dashboard/playground", 
            headers=headers, 
            json=payload, 
            stream=True
        )

        for line in response.iter_lines():
            if line and line.startswith(b"data: "):

                    if data.get("content"):
                        delta["content"] = data["content"]
                        
                    # Reasoning Block (For CoT models like DeepSeek-R1 / O1)
                    if data.get("reasoning"):
                        delta["reasoning"] = data["reasoning"]
                        
                    # Function/Tool Calling
                    if data.get("tool_calls"):
                        delta["tool_calls"] = data["tool_calls"]
                    
                    # Yield the chunk only if it contains actual data
                    if delta:
                        chunk = {
                            "id": f"chatcmpl-{int(time.time())}",
                            "object": "chat.completion.chunk",
                            "model": req_data.get("model", payload["model"]),
                            "choices": [{"index": 0, "delta": delta}]
                        }
                        yield f"data: {json.dumps(chunk)}\n\n"
                    
                    # --- 3. Finalization & Token Usage Stats ---

                                "usage": data["usage"]
                            }
                            yield f"data: {json.dumps(usage_chunk)}\n\n"
                            
                        yield "data: [DONE]\n\n"
                        
                except json.JSONDecodeError:
                    pass

    return Response(generate(), mimetype='text/event-stream')
